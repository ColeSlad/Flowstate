cmake_minimum_required(VERSION 3.24)
project(flowstate VERSION 0.1.0 LANGUAGES CXX)

option(FLOWSTATE_ENABLE_CUDA "Build the CUDA backend (requires a CUDA toolkit)" OFF)
set(FLOWSTATE_SANITIZER "" CACHE STRING "Sanitizer: address, undefined, or thread")
set_property(CACHE FLOWSTATE_SANITIZER PROPERTY STRINGS "" address undefined thread)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(FLOWSTATE_SANITIZER)
  if(NOT FLOWSTATE_SANITIZER MATCHES "^(address|undefined|thread)$")
    message(FATAL_ERROR "Unsupported FLOWSTATE_SANITIZER")
  endif()
  if(FLOWSTATE_ENABLE_CUDA)
    message(FATAL_ERROR "Use a CPU-only build for sanitizers")
  endif()
  add_compile_options(-fsanitize=${FLOWSTATE_SANITIZER} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${FLOWSTATE_SANITIZER})
endif()

add_library(flowstate src/query.cpp src/backends/scalar.cpp src/backends/avx2.cpp
  src/runtime/runtime.cpp src/runtime/scheduler.cpp src/telemetry/metrics.cpp
  src/telemetry/system.cpp src/policy/heuristic.cpp src/workload.cpp)
find_package(Threads REQUIRED)
target_include_directories(flowstate PUBLIC include)
target_link_libraries(flowstate PUBLIC Threads::Threads)
target_compile_options(flowstate PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>)

if(FLOWSTATE_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA requested but nvcc was not found. Use -DFLOWSTATE_ENABLE_CUDA=OFF for CPU-only.")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(flowstate PRIVATE src/backends/cuda.cu)
  target_link_libraries(flowstate PUBLIC CUDA::cudart)
  find_path(FLOWSTATE_NVML_INCLUDE nvml.h HINTS ${CUDAToolkit_INCLUDE_DIRS})
  find_library(FLOWSTATE_NVML_LIBRARY NAMES nvidia-ml libnvidia-ml.so.1 HINTS /usr/lib/wsl/lib)
  if(FLOWSTATE_NVML_INCLUDE AND FLOWSTATE_NVML_LIBRARY)
    target_include_directories(flowstate PRIVATE ${FLOWSTATE_NVML_INCLUDE})
    target_link_libraries(flowstate PRIVATE ${FLOWSTATE_NVML_LIBRARY})
    target_compile_definitions(flowstate PRIVATE FLOWSTATE_HAVE_NVML)
  else()
    message(STATUS "NVML unavailable: GPU utilization/memory telemetry will be absent")
  endif()
  set_target_properties(flowstate PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(flowstate PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
else()
  target_sources(flowstate PRIVATE src/backends/cuda_stub.cpp)
endif()

add_executable(flowstate_search src/main.cpp)
target_link_libraries(flowstate_search PRIVATE flowstate)
add_executable(flowstate_bench benchmarks/search_bench.cpp)
target_link_libraries(flowstate_bench PRIVATE flowstate)
add_executable(flowstate_runtime_bench benchmarks/runtime_bench.cpp)
target_link_libraries(flowstate_runtime_bench PRIVATE flowstate)
add_executable(flowstate_load benchmarks/load_bench.cpp)
target_link_libraries(flowstate_load PRIVATE flowstate)
include(CTest)
if(BUILD_TESTING)
  add_executable(flowstate_scheduler_tests tests/scheduler_tests.cpp)
  target_link_libraries(flowstate_scheduler_tests PRIVATE flowstate)
  add_test(NAME scheduler COMMAND flowstate_scheduler_tests)
  set_tests_properties(scheduler PROPERTIES TIMEOUT 60)
  add_executable(flowstate_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(flowstate_runtime_tests PRIVATE flowstate)
  add_test(NAME runtime COMMAND flowstate_runtime_tests)
  add_test(NAME runtime_gpu COMMAND flowstate_runtime_tests gpu)
  set_tests_properties(runtime runtime_gpu PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
  add_executable(flowstate_tests tests/search_tests.cpp)
  target_link_libraries(flowstate_tests PRIVATE flowstate)
  foreach(backend scalar avx2 cuda)
    add_test(NAME search_${backend} COMMAND flowstate_tests ${backend})
    set_tests_properties(search_${backend} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
  endforeach()
  add_test(NAME cli_smoke COMMAND flowstate_search --vectors 100 --dimension 9 --top-k 3)
  add_test(NAME cli_invalid COMMAND ${CMAKE_COMMAND}
    -DPROGRAM=$<TARGET_FILE:flowstate_search> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_invalid.cmake)
  set_tests_properties(cli_smoke cli_invalid PROPERTIES TIMEOUT 60)
endif()
